Presioná "Ejecutar δ" para ver el recorrido nodo por nodo.
Pseudocódigo
// Recorre la lista buscando (from==q && symbol==a)
Tdata delta(Automata* A, State q, Symbol a) {
  TransitionNode* curr = A->Delta;
  while (curr != NULL) {
    if (curr->t.from == q && curr->t.symbol == a)
      return curr->t.to;  // SET de destinos
    curr = curr->next;
  }
  return NULL;  // transición no definida
}
